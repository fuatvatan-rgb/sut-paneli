// Sayı çevirici — "1.234,56" ve "1234,56" biçimlerini de tanır function sayi(v) { if (v === null || v === undefined || v === "") return NaN; let s = String(v).trim(); if (s.includes(",")) s = s.replace(/\./g, "").replace(",", "."); return parseFloat(s); } // Ağırlıklı ortalaması alınacak analiz kolonları const ANALIZ = ["Sıcaklık", "Ph", "Sh", "Yağ", "Klasik Yöntem", "Protein", "Laktoz", "TKM", "YKM", "Donma Noktası", "Somatik", "Yoğunluk", "Su", "Laktam", "Alkol", "Koku", "Test"]; // Aynı seferi tanımlayan kolonlar — Excel'inde adları farklıysa burayı düzelt function seferKey(r) { return [r["Plaka"], r["Tarih"], r["Saat"], r["Sefer Adı"], r["Mühür No"]].join("|"); } function seferBirlestir(satirlar) { const gruplar = new Map(); for (const r of satirlar) { const k = seferKey(r); if (!gruplar.has(k)) gruplar.set(k, []); gruplar.get(k).push(r); } const sonuc = []; for (const gozler of gruplar.values()) { const ilk = gozler[0]; const kayit = {}; // Sefer sabit bilgileri — ilk satırdan for (const kolon of ["Tarih", "Saat", "Plaka", "Araç", "Süt Cinsi", "Sefer Adı", "Mühür No", "Kullanıcı"]) { kayit[kolon] = ilk[kolon]; } // MİKTAR: tanker kantar değeri SADECE BİR KEZ alınır kayit["Miktar"] = sayi(ilk["Kantar"]); kayit["Göz Sayısı"] = gozler.length; // Göz ağırlıkları const agirliklar = gozler.map(function (g) { const m = sayi(g["Göz Miktarı"]); return isNaN(m) ? 0 : m; }); const agirlikToplam = agirliklar.reduce(function (a, b) { return a + b; }, 0); // ANALİZLER: ağırlıklı ortalama for (const kolon of ANALIZ) { let pay = 0, payda = 0; gozler.forEach(function (g, i) { const deger = sayi(g[kolon]); if (isNaN(deger)) return; // Göz miktarı yoksa eşit ağırlık kullanılır const w = agirlikToplam > 0 ? agirliklar[i] : 1; pay += deger * w; payda += w; }); kayit[kolon] = payda > 0 ? +(pay / payda).toFixed(3) : null; } sonuc.push(kayit); } return sonuc; }
